## Usage

```tsx showLineNumbers
import {
  Pagination,
  PaginationEllipsis,
  PaginationItem,
  PaginationItems,
  PaginationNext,
  PaginationPrevious
} from "~/components/ui/pagination"
```

```tsx showLineNumbers
<Pagination
  itemComponent={(props) => <PaginationItem page={props.page}>{props.page}</PaginationItem>}
  ellipsisComponent={() => <PaginationEllipsis />}
>
  <PaginationPrevious />
  <PaginationItems />
  <PaginationNext />
</Pagination>
```
