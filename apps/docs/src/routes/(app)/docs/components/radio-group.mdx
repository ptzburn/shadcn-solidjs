## Usage

```tsx
import { RadioGroup, RadioGroupItem, RadioGroupItemLabel } from "~/components/ui/radio-group"
```

```tsx
<RadioGroup>
  <For each={["Apple", "Orange", "Watermelon"]}>
    {(fruit) => (
      <RadioGroupItem value={fruit}>
        <RadioGroupItemLabel>{fruit}</RadioGroupItemLabel>
      </RadioGroupItem>
    )}
  </For>
</RadioGroup>
```
