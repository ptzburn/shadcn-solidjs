## Controlled

Use the `pressed` and `onChange` props to control the toggle.

```tsx showLineNumbers
import { createSignal } from "solid-js"

import { Toggle } from "~/components/ui/toggle"

export function ToggleControlled() {
  const [pressed, setPressed] = createSignal(false)

  return (
    <Toggle pressed={pressed()} onChange={setPressed} aria-label="Toggle bold">
      Bold
    </Toggle>
  )
}
```

The children can also be a render prop that receives the toggle state.

```tsx showLineNumbers
<Toggle aria-label="Toggle bold">
  {(state) => (
    <Show when={state.pressed()} fallback={<IconOff />}>
      <IconOn />
    </Show>
  )}
</Toggle>
```
